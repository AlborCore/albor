# the hello-world example, which demonstrates the usage of the framework with different backends (§6.3, AD-26)

# the example drives a desktop platform directly, so it only builds where a renderer umbrella exists (AD-16)
# the umbrella is the public target the top level creates for one platform and renderer pair (§3.3, AD-19)
if(NOT TARGET albor AND NOT TARGET albor-blend2d)
    message(STATUS "hello-world: skipped, the example needs a desktop platform and a renderer")
    return()
endif()

# gather cpp sources (all platforms)
file(GLOB_RECURSE CPP_SOURCES "./*.cpp")

# the recommended renderer is the umbrella, which pairs the platform with the backend it chose (§3.3, AD-19)
if(TARGET albor)
    add_executable(hello-world-albor ${CPP_SOURCES})
    # the example chooses only the renderer; the public target carries the rest (§3.3, AD-19)
    target_include_directories(hello-world-albor PRIVATE ${ALBOR_SOURCE_DIR})
    target_compile_features(hello-world-albor PRIVATE cxx_std_20)
    target_link_libraries(hello-world-albor PRIVATE albor)
    # when automation is enabled, link the whole-archive umbrella which installs the host at load (§10.3, AD-10)
    if(ALBOR_AUTOMATION)
        target_link_libraries(hello-world-albor PRIVATE albor-automation-auto)
    endif()
endif()

# the portable cpu renderer is a separate choice, which exists only where blend2d was configured (§6.3, AD-26)
if(TARGET albor-blend2d)
    add_executable(hello-world-blend2d ${CPP_SOURCES})
    # the example chooses only the renderer; the public target carries the rest (§3.3, AD-19)
    target_include_directories(hello-world-blend2d PRIVATE ${ALBOR_SOURCE_DIR})
    target_compile_features(hello-world-blend2d PRIVATE cxx_std_20)
    target_link_libraries(hello-world-blend2d PRIVATE albor-blend2d)
    # when automation is enabled, link the whole-archive umbrella which installs the host at load (§10.3, AD-10)
    if(ALBOR_AUTOMATION)
        target_link_libraries(hello-world-blend2d PRIVATE albor-automation-auto)
    endif()
endif()
